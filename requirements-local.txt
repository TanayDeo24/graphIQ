# Extra dependencies for LLM_BACKEND=local (Apple Silicon MLX dev backend)
# ONLY -- not needed, and not installable, on Render or any other
# non-macOS deployment target, which always runs LLM_BACKEND=cloudflare.
# Install alongside the base requirements for local development:
#   pip install -r requirements.txt -r requirements-local.txt
#
# Split into its own file (previously these three lines carried a
# `sys_platform == "darwin"` marker directly in requirements.txt) so a
# platform-marker mismatch or an unconditional install command can never
# accidentally pull these into a Linux deploy in the first place -- the
# code-level conditional import in src/agent/llm_backend.py (which never
# imports src/agent/local_backend.py, and therefore never imports mlx,
# unless LLM_BACKEND=local) is the real guarantee; this file split is a
# second, independent layer of the same protection.
mlx>=0.20,<1
mlx-lm>=0.20,<1
# Local embeddings for doc retrieval when LLM_BACKEND=local. NOT
# mlx-embeddings -- that package forces numpy>=2, directly incompatible
# with lifelines/scikit-survival's numpy<2 requirement (hit live: pip
# force-upgraded numpy to 2.4.6 and broke lifelines on first install
# attempt). sentence-transformers runs BGE on PyTorch (already a project
# dependency for the spend autoencoder) and stays numpy<2 compatible.
sentence-transformers>=3.0,<7
